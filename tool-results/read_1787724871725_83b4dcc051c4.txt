   330→        </div>
   331→        {kpi.change !== 0 && (
   332→          <span className={`inline-flex items-center px-2 py-0.5 rounded-full text-xs font-semibold ${
   333→            isPositive 
   334→              ? 'bg-emerald-100 text-emerald-700 border border-emerald-200'

   336→          }`}>

   339→          </span>
   340→        )}
   341→      </div>
   342→
   343→      <div className="relative z-10">
   344→        <p className="text-xs font-medium text-slate-500 mb-1">{kpi.titleAr || kpi.titleKey}</p>
   345→        <h3 className="text-xl font-bold text-slate-900">{kpi.value}</h3>
   346→      </div>
   347→    </div>
   348→  )
   349→}
